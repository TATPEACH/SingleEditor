cmake_minimum_required(VERSION 3.22)
project(SingleEditorDiaInspector LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This build-side inspector requires MSVC x64")
endif()
set(SINGLE_EDITOR_DIA_ROOT "" CACHE PATH "Explicit installed Visual Studio DIA SDK")
if(NOT EXISTS "${SINGLE_EDITOR_DIA_ROOT}/include/dia2.h")
    message(FATAL_ERROR "The installed DIA SDK is required")
endif()
add_executable(SingleEditorDiaInspector DiaSectionContributions.cpp)
target_compile_features(SingleEditorDiaInspector PRIVATE cxx_std_20)
target_compile_definitions(SingleEditorDiaInspector PRIVATE UNICODE _UNICODE NOMINMAX)
target_include_directories(SingleEditorDiaInspector PRIVATE "${SINGLE_EDITOR_DIA_ROOT}/include")
target_link_libraries(SingleEditorDiaInspector PRIVATE
    "${SINGLE_EDITOR_DIA_ROOT}/lib/amd64/diaguids.lib" ole32 oleaut32)

cmake_minimum_required(VERSION 3.24)
project(SingleEditorEtlOfflineTools LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "These isolated ToolTestOnly readers require Windows x64")
endif()
add_executable(SingleEditorEtlDecode SingleEditorEtlDecode.cpp)
add_executable(SingleEditorEtlDecodeContractTests DecoderContractTests.cpp)
foreach(target IN ITEMS SingleEditorEtlDecode SingleEditorEtlDecodeContractTests)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /EHsc)
  target_link_libraries(${target} PRIVATE tdh advapi32 bcrypt)
endforeach()
# Intentionally no install(), shipping target, product linkage, collector or JUCE.

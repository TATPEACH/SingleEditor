cmake_minimum_required(VERSION 3.24)
project(SingleEditorWavMetadataTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${SINGLE_EDITOR_JUCE_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Set SINGLE_EDITOR_JUCE_ROOT to the reviewed local JUCE source")
endif()
add_subdirectory("${SINGLE_EDITOR_JUCE_ROOT}" "${CMAKE_BINARY_DIR}/juce")
include(../../SingleEditorJuceWavMetadata.cmake)
single_editor_configure_juce_wav_metadata()
add_executable(SingleEditorWavMetadataTests ../../../Tests/WavMetadataTests.cpp)
target_compile_definitions(SingleEditorWavMetadataTests PRIVATE
    JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1
    JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
target_link_libraries(SingleEditorWavMetadataTests PRIVATE juce::juce_audio_formats)
if(MSVC)
    target_compile_options(SingleEditorWavMetadataTests PRIVATE /bigobj /Gy)
endif()
enable_testing()
add_test(NAME wav-metadata-roundtrip COMMAND SingleEditorWavMetadataTests)
